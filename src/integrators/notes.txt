# uniform_sample_one_light() gets updated to have a final arg which is a light distribution which is a distribution1D
# uniform_sample_one_light is just like (sample light via estimate direct) / (prob of selecting light to be sampled)
# So light distribution would have several potential stratigies. 
    1. uniform
    2. power (not helpful)
    3. spatial (hard)
    4. distance (john's idea)

p = inter.core.p
inv_distances = Vector{Float64}(undef, length(lights))
for i, light in enumerate(lights)
	if light isa AreaLight
		d = typemax(Float64)
		for corner in corners(light.box)
			dd = distance(p, corner)
			if dd < d
				d = dd
			end
		end
	else
		d = distance(p, light.center)
	end
	distances[i] = 1/d
end

return Distribution1D(distances)
